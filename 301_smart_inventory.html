โค้ด Google Apps Script (สำหรับ รพ.สต. บัตร 301 & User) 

// Google Apps Script สำหรับ รพ.สต. (Anti-Negative, Min/Max, 301 Cardex, User)
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var action = e ? e.parameter.action : '';
  
  if (action === 'initDB') {
    return initDatabase(ss);
  } else if (action === 'getData') {
    return handleGetData(ss);
  } else if (action === 'receive') {
    return handleReceive(e, ss);
  } else if (action === 'issue') {
    return handleIssue(e, ss);
  } else if (action === 'history') {
    return handleHistory(ss);
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success", message: "RPST Pharmacy API Ready"}))
         .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  return doGet(e);
}

function initDatabase(ss) {
  var stockSheet = ss.getSheetByName("Stock");
  if (!stockSheet) {
    stockSheet = ss.insertSheet("Stock");
    stockSheet.appendRow(["Code", "Name", "Lot", "ExpDate", "ReceiveDate", "Qty", "Category", "Source", "Min", "Max"]);
    stockSheet.appendRow(["MED001", "Paracetamol 500mg", "LOT2026-01", "2027-12-31", "2026-01-10", 2000, "ยาสามัญประจำบ้าน", "องค์การเภสัชกรรม", 100, 5000]);
    stockSheet.appendRow(["MED002", "Amoxicillin 500mg", "LOT2026-02", "2026-08-15", "2026-01-12", 1000, "ยาอันตราย", "องค์การเภสัชกรรม", 50, 2000]);
    stockSheet.appendRow(["SUP001", "Syringe 3ml with Needle", "LOT2026-S1", "2028-05-20", "2026-01-15", 500, "เวชภัณฑ์มิใช่ยา", "บริษัท ไทยเมดิพลัส", 100, 1000]);
  }
  
  var histSheet = ss.getSheetByName("History");
  if (!histSheet) {
    histSheet = ss.insertSheet("History");
    histSheet.appendRow(["Type", "Code", "Name", "Lot", "Qty", "Date", "Receiver", "User"]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Database initialized successfully"}))
         .setMimeType(ContentService.MimeType.JSON);
}

function handleGetData(ss) {
  var sheet = ss.getSheetByName("Stock") || ss.getSheets()[0];
  var rows = sheet.getDataRange().getValues();
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    data.push({
      code: rows[i][0],
      name: rows[i][1],
      lot: rows[i][2],
      exp: rows[i][3],
      receiveDate: rows[i][4],
      qty: rows[i][5],
      category: rows[i][6] || 'ยาสามัญประจำบ้าน',
      source: rows[i][7] || '-',
      min: rows[i][8] !== undefined ? rows[i][8] : 10,
      max: rows[i][9] !== undefined ? rows[i][9] : 1000
    });
  }
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function handleReceive(e, ss) {
  var sheet = ss.getSheetByName("Stock") || ss.getSheets()[0];
  sheet.appendRow([
    e.parameter.code, 
    e.parameter.name, 
    e.parameter.lot, 
    e.parameter.exp, 
    e.parameter.date, 
    parseInt(e.parameter.qty), 
    e.parameter.category, 
    e.parameter.source,
    parseInt(e.parameter.min || 10),
    parseInt(e.parameter.max || 1000)
  ]);
  
  var histSheet = ss.getSheetByName("History") || ss.insertSheet("History");
  if (histSheet.getLastRow() === 0) histSheet.appendRow(["Type", "Code", "Name", "Lot", "Qty", "Date", "Receiver", "User"]);
  histSheet.appendRow(["RECEIVE", e.parameter.code, e.parameter.name, e.parameter.lot, parseInt(e.parameter.qty), e.parameter.date, e.parameter.source, e.parameter.user || 'System']);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}

function handleIssue(e, ss) {
  var sheet = ss.getSheetByName("Stock") || ss.getSheets()[0];
  var rows = sheet.getDataRange().getValues();
  var code = e.parameter.code;
  var lot = e.parameter.lot;
  var qtyToIssue = parseInt(e.parameter.qty);
  var found = false;
  
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0].toString() === code.toString() && rows[i][2].toString() === lot.toString()) {
      var currentQty = parseInt(rows[i][5]);
      
      // Anti-Negative Balance Guard
      if (currentQty < qtyToIssue) {
        return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Insufficient stock"}))
               .setMimeType(ContentService.MimeType.JSON);
      }
      
      var newQty = currentQty - qtyToIssue;
      sheet.getRange(i + 1, 6).setValue(newQty);
      found = true;
      
      var histSheet = ss.getSheetByName("History") || ss.insertSheet("History");
      histSheet.appendRow(["ISSUE", rows[i][0], rows[i][1], lot, qtyToIssue, new Date().toISOString().split('T')[0], e.parameter.receiver, e.parameter.user || 'System']);
      break;
    }
  }
  return ContentService.createTextOutput(JSON.stringify({status: found ? "success" : "not_found"})).setMimeType(ContentService.MimeType.JSON);
}

function handleHistory(ss) {
  var sheet = ss.getSheetByName("History");
  if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  var rows = sheet.getDataRange().getValues();
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    data.push({
      type: rows[i][0],
      code: rows[i][1],
      name: rows[i][2],
      lot: rows[i][3],
      qty: rows[i][4],
      date: rows[i][5],
      receiver: rows[i][6],
      user: rows[i][7] || '-'
    });
  }
  return ContentService.createTextOutput(JSON.stringify(data.reverse())).setMimeType(ContentService.MimeType.JSON);
}
