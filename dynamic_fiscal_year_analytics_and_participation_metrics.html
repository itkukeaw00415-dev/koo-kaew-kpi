/**
 * KUKAEW HEALTH CHALLENGE 2570 - Auto-Provisioning Google Apps Script
 * เพียงวางโค้ดนี้ แล้ว Deploy > New deployment > Web app (Who has access: Anyone)
 * เมื่อนำลิงก์มาวางในเว็บแอป ระบบจะสร้างชีตฐานข้อมูลทั้งหมดให้อัตโนมัติทันที!
 */
function setupDatabase() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetsConfig = {
    'VHV_MASTER': ['VHVCode', 'OriginalNo', 'FullName', 'Subdistrict', 'VillageCode', 'VillageNo', 'VillageName', 'ServiceUnit', 'Phone', 'Active', 'Remark'],
    'USERS': ['Email', 'FullName', 'Role', 'VHVCode', 'VillageCode', 'Subdistrict', 'Active'],
    'VILLAGE_MASTER': ['VillageCode', 'Subdistrict', 'VillageNo', 'VillageName', 'TotalVHV'],
    'IMPORT_CHECK': ['CheckItem', 'Target', 'Actual', 'Diff', 'Status'],
    'ACTIVITY_LOG': ['ActivityID', 'Timestamp', 'VHVCode', 'FullName', 'Subdistrict', 'VillageCode', 'Category', 'Detail', 'PhotoData', 'Points', 'Status', 'ApprovedBy'],
    'SCORE_LEDGER': ['LedgerID', 'Timestamp', 'VHVCode', 'Subdistrict', 'VillageCode', 'ActivityID', 'Points', 'Remark']
  };

  let createdSheets = [];
  Object.keys(sheetsConfig).forEach(sheetName => {
    let sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      createdSheets.push(sheetName);
    }
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(sheetsConfig[sheetName]);
      sheet.getRange(1, 1, 1, sheetsConfig[sheetName].length)
        .setFontWeight('bold')
        .setBackground('#0F4C5C')
        .setFontColor('#FFFFFF');
      sheet.setFrozenRows(1);
    }
  });

  // เติมข้อมูลตั้งต้นอัตโนมัติหาก VHV_MASTER ยังว่างอยู่
  const vhvSheet = ss.getSheetByName('VHV_MASTER');
  if (vhvSheet && vhvSheet.getLastRow() === 1) {
    const seedVhv = [
      ['BJ-VHV-0001', 1, 'นาง สมพร ศรีบุรินทร์', 'บ้านจีต', 'BJ-01', 1, 'บ้านจีต', 'รพ.สต.บ้านจีต', '', true, 'ตัวอย่างตั้งต้น'],
      ['KS-VHV-0001', 1, 'นาง อารยา อินทะนู', 'คอนสาย', 'KS-01', 1, 'บ้านคอนสาย', 'PCU รพ.กู่แก้ว', '', true, 'ตัวอย่างตั้งต้น'],
      ['KY-VHV-0001', 1, 'นาง สำราญ แก้วมณี', 'ค้อใหญ่', 'KY-01', 1, 'บ้านค้อใหญ่', 'รพ.สต.ค้อใหญ่', '', true, 'ตัวอย่างตั้งต้น'],
      ['NT-VHV-0001', 1, 'นาง สงบ บุญทัน', 'โนนทองอินทร์', 'NT-01', 1, 'บ้านยาง', 'รพ.สต.โนนทองอินทร์', '', true, 'ตัวอย่างตั้งต้น']
    ];
    seedVhv.forEach(r => vhvSheet.appendRow(r));
  }

  // เติมสูตรตรวจสอบอัตโนมัติใน IMPORT_CHECK หากยังว่างอยู่
  const checkSheet = ss.getSheetByName('IMPORT_CHECK');
  if (checkSheet && checkSheet.getLastRow() === 1) {
    const checks = [
      ['จำนวนรายชื่อ อสม. ทั้งหมด (คน)', 480, '=COUNTA(VHV_MASTER!A2:A481)', '=C2-B2', '=IF(C2=B2,"ผ่าน","อยู่ระหว่างนำเข้า")'],
      ['ตำบลบ้านจีต (BJ)', 135, '=COUNTIF(VHV_MASTER!D2:D481,"บ้านจีต")', '=C3-B3', '=IF(C3=B3,"ผ่าน","อยู่ระหว่างนำเข้า")'],
      ['ตำบลคอนสาย (KS)', 153, '=COUNTIF(VHV_MASTER!D2:D481,"คอนสาย")', '=C4-B4', '=IF(C4=B4,"ผ่าน","อยู่ระหว่างนำเข้า")'],
      ['ตำบลค้อใหญ่ (KY)', 81, '=COUNTIF(VHV_MASTER!D2:D481,"ค้อใหญ่")', '=C5-B5', '=IF(C5=B5,"ผ่าน","อยู่ระหว่างนำเข้า")'],
      ['ตำบลโนนทองอินทร์ (NT)', 111, '=COUNTIF(VHV_MASTER!D2:D481,"โนนทองอินทร์")', '=C6-B6', '=IF(C6=B6,"ผ่าน","อยู่ระหว่างนำเข้า")']
    ];
    checks.forEach(r => checkSheet.appendRow(r));
  }

  return createdSheets;
}

function doGet(e) {
  // สร้างฐานข้อมูลอัตโนมัติทันทีหากยังไม่มีชีตครบ
  const created = setupDatabase();

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const vhvSheet = ss.getSheetByName('VHV_MASTER');
  const actSheet = ss.getSheetByName('ACTIVITY_LOG');
  const ledgerSheet = ss.getSheetByName('SCORE_LEDGER');

  const vhvData = vhvSheet ? sheetToObjects(vhvSheet) : [];
  const actData = actSheet ? sheetToObjects(actSheet) : [];
  const ledgerData = ledgerSheet ? sheetToObjects(ledgerSheet) : [];

  const scoreMap = {};
  ledgerData.forEach(item => {
    const code = String(item.VHVCode || '').trim().toUpperCase();
    if (code) scoreMap[code] = (scoreMap[code] || 0) + (Number(item.Points) || 0);
  });

  const enrichedVhv = vhvData.map(v => {
    const code = String(v.VHVCode || '').trim().toUpperCase();
    v.Points = scoreMap[code] || 0;
    return v;
  });

  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    autoInitialized: created.length > 0,
    createdSheets: created,
    vhv: enrichedVhv,
    activities: actData
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  setupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const payload = JSON.parse(e.postData.contents);
  const vhvSheet = ss.getSheetByName('VHV_MASTER');
  const actSheet = ss.getSheetByName('ACTIVITY_LOG');
  const ledgerSheet = ss.getSheetByName('SCORE_LEDGER');

  if (payload.action === 'INIT_DB') {
    const created = setupDatabase();
    return jsonResponse({ status: 'ok', createdSheets: created });
  }

  if (payload.action === 'ADD_ACTIVITY') {
    const id = 'ACT-' + new Date().getTime();
    actSheet.appendRow([
      id, new Date(), payload.vhvCode, payload.fullName,
      payload.subdistrict, payload.villageCode, payload.category,
      payload.detail, payload.photoData || '', payload.points, 'PENDING', ''
    ]);
    return jsonResponse({ status: 'ok', activityId: id });
  }

  if (payload.action === 'APPROVE_ACTIVITY') {
    const rows = actSheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(payload.activityId) && rows[i][10] !== 'APPROVED') {
        actSheet.getRange(i + 1, 11).setValue('APPROVED');
        actSheet.getRange(i + 1, 12).setValue(payload.approvedBy || 'DISTRICT');
        ledgerSheet.appendRow([
          'LED-' + new Date().getTime(), new Date(),
          rows[i][2], rows[i][4], rows[i][5], rows[i][0], rows[i][9], 'อนุมัติคะแนนกิจกรรม'
        ]);
        break;
      }
    }
    return jsonResponse({ status: 'ok' });
  }

  if (payload.action === 'UPDATE_ACTIVITY') {
    const rows = actSheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(payload.activityId)) {
        actSheet.getRange(i + 1, 7).setValue(payload.category);
        actSheet.getRange(i + 1, 8).setValue(payload.detail);
        actSheet.getRange(i + 1, 9).setValue(Number(payload.points));
        actSheet.getRange(i + 1, 10).setValue(payload.status);
        break;
      }
    }
    return jsonResponse({ status: 'ok' });
  }

  if (payload.action === 'DELETE_ACTIVITY') {
    const rows = actSheet.getDataRange().getValues();
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(payload.activityId)) {
        actSheet.deleteRow(i + 1);
        break;
      }
    }
    return jsonResponse({ status: 'ok' });
  }

  if (payload.action === 'UPSERT_VHV') {
    const rows = vhvSheet.getDataRange().getValues();
    let updated = false;
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]).trim().toUpperCase() === String(payload.vhvCode).trim().toUpperCase()) {
        vhvSheet.getRange(i + 1, 3).setValue(payload.fullName);
        vhvSheet.getRange(i + 1, 4).setValue(payload.subdistrict);
        vhvSheet.getRange(i + 1, 5).setValue(payload.villageCode);
        vhvSheet.getRange(i + 1, 6).setValue(payload.villageNo || '');
        vhvSheet.getRange(i + 1, 7).setValue(payload.villageName || '');
        vhvSheet.getRange(i + 1, 8).setValue(payload.serviceUnit || '');
        vhvSheet.getRange(i + 1, 9).setValue(payload.phone || '');
        vhvSheet.getRange(i + 1, 10).setValue(payload.active !== false);
        updated = true;
        break;
      }
    }
    if (!updated) {
      vhvSheet.appendRow([
        payload.vhvCode, rows.length, payload.fullName, payload.subdistrict,
        payload.villageCode, payload.villageNo || '', payload.villageName || '',
        payload.serviceUnit || '', payload.phone || '', true, payload.remark || ''
      ]);
    }
    return jsonResponse({ status: 'ok' });
  }

  if (payload.action === 'DELETE_VHV') {
    const rows = vhvSheet.getDataRange().getValues();
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]).trim().toUpperCase() === String(payload.vhvCode).trim().toUpperCase()) {
        vhvSheet.deleteRow(i + 1);
        break;
      }
    }
    return jsonResponse({ status: 'ok' });
  }
}

function sheetToObjects(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  const headers = data[0].map(h => String(h).trim());
  return data.slice(1).filter(r => String(r[0]).trim() !== '').map(row => {
    const obj = {};
    headers.forEach((h, idx) => obj[h] = row[idx]);
    return obj;
  });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
